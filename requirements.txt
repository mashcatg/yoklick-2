-- Table: thumbnails (each image = one row)
create table if not exists thumbnails (
  id bigserial primary key,
  created_at timestamp with time zone default now(),
  prompt text,
  image_url text,
  user_id uuid references auth.users(id) on delete cascade,
  text_position text,
  is_public boolean default false,
  is_featured boolean default false,
  group_id uuid
);

-- Table: users_meta (credits and plan info)
create table if not exists users_meta (
  id uuid primary key references auth.users(id) on delete cascade,
  credits int default 0,
  plan text default 'free', -- free, starter, pro, agency
  is_yearly boolean default false,
  created_at timestamp with time zone default now()
);

-- Table: transactions (payment log)
create table if not exists transactions (
  id uuid primary key default gen_random_uuid(),
  user_id uuid references auth.users(id) on delete cascade,
  amount numeric not null,
  credits int not null,
  plan text,
  is_yearly boolean,
  payment_provider text default 'dodo',
  created_at timestamp with time zone default now()
);

-- Table: feedback (user input)
create table if not exists feedback (
  id uuid primary key default gen_random_uuid(),
  user_id uuid references auth.users(id) on delete cascade,
  message text not null,
  created_at timestamp with time zone default now()
);

export default function SideBar({ className = "" }: { className?: string }) {
  const pathname = usePathname()
  const [credits, setCredits] = useState<number | null>(null)
  const [isLoading, setIsLoading] = useState(true)

  useEffect(() => {
    const fetchCredits = async () => {
      const { data: { session } } = await supabase.auth.getSession()
      if (!session) {
        setIsLoading(false)
        return
      }

      const { data, error } = await supabase
        .from('user_credits')
        .select('balance')
        .eq('user_id', session.user.id)
        .single()

      if (!error && data) {
        setCredits(data.balance)
      }
      setIsLoading(false)

      // Subscribe to credits changes
      const channel = supabase
        .channel('credits')
        .on('postgres_changes', {
          event: '*',
          schema: 'public',
          table: 'user_credits',
          filter: `user_id=eq.${session.user.id}`
        }, () => {
          fetchCredits()
        })
        .subscribe()

      return () => {
        channel.unsubscribe()
      }
    }

    fetchCredits()
  }, [])

  if (isLoading) {
    return <div className="bg-muted/50 rounded-lg p-3 animate-pulse h-8" />
  }
  
  const isActive = (href: string) => pathname === href

  return (
    <aside
      className={`w-16 fixed min-h-[calc(100vh-4rem)] dark:bg-[#303030] mx-2 flex flex-col items-center py-4 z-50 ${className}`}
    >
      <div className="flex flex-col items-center gap-3">
        {nav.map(({ href, label, Icon }) => {
          const active = isActive(href)

      <div className="mt-auto flex flex-col items-center gap-4 mb-2">
        {/* Credits Link */}
